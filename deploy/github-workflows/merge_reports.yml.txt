# 合并报告推送 —— 独立 workflow，與定时抓取完全独立。
# 一对一：合并后只发给自己（token 所属账号）
name: 章鱼AI 合并报告推送

on:
  workflow_dispatch:
    inputs:
      files:
        description: "待合并的 Markdown 文件（逗号分隔，如 goldwind_analysis.md,multi_factor_report.md）"
        required: true
        default: "goldwind_analysis.md,multi_factor_report.md"
        type: string
      merge_topic:
        description: "合并后主题标题（可选）"
        required: false
        default: "多因子量化模型与金风科技实证 - 合并报告"
        type: string
      dry_run:
        description: "只生成预览不推送"
        required: false
        default: "false"
        type: choice
        options:
          - "false"
          - "true"

permissions:
  contents: write

jobs:
  merge-push:
    runs-on: ubuntu-latest
    timeout-minutes: 5

    steps:
      - name: 检出代码
        uses: actions/checkout@v5

      - name: 安装 Python
        uses: actions/setup-python@v6
        with:
          python-version: "3.11"
          cache: pip

      - name: 安装依赖
        run: pip install -r requirements.txt

      - name: 合并报告并推送
        env:
          PUSHPLUS_TOKEN: ${{ secrets.PUSHPLUS_TOKEN }}
          DEEPSEEK_API_KEY: ${{ secrets.DEEPSEEK_API_KEY }}
          ZHIPU_API_KEY: ${{ secrets.ZHIPU_API_KEY }}
          DEEPSEEK_MODEL: ${{ secrets.DEEPSEEK_MODEL }}
          TZ: Asia/Shanghai
        run: |
          echo "合并文件：${{ inputs.files }}"
          if [ "${{ inputs.dry_run }}" = "true" ]; then
            python main.py --merge ${{ inputs.files }} --merge-topic "${{ inputs.merge_topic }}" --dry-run
          else
            python main.py --merge ${{ inputs.files }} --merge-topic "${{ inputs.merge_topic }}"
          fi

      - name: 生成示范合并报告产物
        if: always()
        run: |
          python main.py --generate-merged-report --dry-run
          ls -lh merged_report.md

      - name: 上传预览/合并报告
        if: always()
        uses: actions/upload-artifact@v6
        with:
          name: merged-report
          path: |
            preview.html
            merged_report.md
          if-no-files-found: ignore
          retention-days: 3
